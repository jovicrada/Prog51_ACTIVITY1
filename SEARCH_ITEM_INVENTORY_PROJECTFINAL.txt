#include <iostream>
using namespace std;

int main() {

    int choice;

    string Item_name[100];
    int Price[100], Quantity[100];
    int count = 0;
    int found;

    do {

        cout << "\n===== INVENTORY MENU =====\n";
        cout << "1. Add Item\n";
        cout << "2. View All Items\n";
        cout << "3. Search Item\n";
        cout << "4. Update Item\n";
        cout << "5. Delete Item\n";
        cout << "6. Inventory Report\n";
        cout << "7. Exit\n";

        cout << "\nENTER CHOICE: ";
        cin >> choice;

        switch (choice) {

             case 3: {
                string search;

                cout << "Enter item name: ";
                cin >> search;

                found = 0;

                for (int i = 0; i < count; i++) {

                    if (Item_name[i] == search) {

                        cout << "\nItem found!";
                        cout << "\nQuantity: " << Quantity[i];
                        cout << "\nPrice: " << Price[i];

                        found = 1;
                        break;
                    }
                }

                if (found == 0) {
                    cout << "Item not found!";
                }

                break;
            }
        
       
    }while (choice != 7);

    return 0;
}
