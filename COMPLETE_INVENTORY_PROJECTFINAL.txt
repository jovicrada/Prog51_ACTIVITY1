#include <iostream>
using namespace std;

int main() {

    int choice;

    string Item_name[100];
    int Price[100], Quantity[100];
    int count = 0;
    int found;

    do {

        cout << "\n===== INVENTORY MENU =====\n";
        cout << "1. Add Item\n";
        cout << "2. View All Items\n";
        cout << "3. Search Item\n";
        cout << "4. Update Item\n";
        cout << "5. Delete Item\n";
        cout << "6. Inventory Report\n";
        cout << "7. Exit\n";

        cout << "\nENTER CHOICE: ";
        cin >> choice;

        switch (choice) {

            case 1: {
                cout << "\nItem name: ";
                cin >> Item_name[count];

                cout << "Quantity: ";
                cin >> Quantity[count];

                cout << "Price: ";
                cin >> Price[count];

                count++;

                cout << "Item added!";
                break;
            }

            
            case 2: {
                for (int i = 0; i < count; i++) {

                    cout << "\nItem name: " << Item_name[i];
                    cout << "\nQuantity: " << Quantity[i];
                    cout << "\nPrice: " << Price[i] << endl;
                }

                if (count == 0) {
                    cout << "No items available!";
                }

                break;
            }

            
            case 3: {
                string search;

                cout << "Enter item name: ";
                cin >> search;

                found = 0;

                for (int i = 0; i < count; i++) {

                    if (Item_name[i] == search) {

                        cout << "\nItem found!";
                        cout << "\nQuantity: " << Quantity[i];
                        cout << "\nPrice: " << Price[i];

                        found = 1;
                        break;
                    }
                }

                if (found == 0) {
                    cout << "Item not found!";
                }

                break;
            }

            
            case 4: {
                string search;

                cout << "Enter item name to update: ";
                cin >> search;

                found = 0;

                for (int i = 0; i < count; i++) {

                    if (Item_name[i] == search) {

                        cout << "New quantity: ";
                        cin >> Quantity[i];

                        cout << "New price: ";
                        cin >> Price[i];

                        cout << "Item updated!";

                        found = 1;
                        break;
                    }
                }

                if (found == 0) {
                    cout << "Item not found!";
                }

                break;
            }

            
            case 5: {
                string search;

                cout << "Enter item name to delete: ";
                cin >> search;

                found = 0;

                for (int i = 0; i < count; i++) {

                    if (Item_name[i] == search) {

                        for (int j = i; j < count - 1; j++) {

                            Item_name[j] = Item_name[j + 1];
                            Quantity[j] = Quantity[j + 1];
                            Price[j] = Price[j + 1];
                        }

                        count--;

                        cout << "Item deleted!";

                        found = 1;
                        break;
                    }
                }

                if (found == 0) {
                    cout << "Item not found!";
                }

                break;
            }

            
            case 6: {
                int Total_Quantity = 0;
                int Total_Value = 0;

                for (int i = 0; i < count; i++) {

                    Total_Quantity += Quantity[i];
                    Total_Value += Quantity[i] * Price[i];
                }

                cout << "\n===== INVENTORY REPORT =====";
                cout << "\nTotal Items: " << count;
                cout << "\nTotal Quantity: " << Total_Quantity;
                cout << "\nTotal Inventory Value: " << Total_Value;

                break;
            }

            
            case 7: {
                cout << "Exiting Inventory System...";
                break;
            }

            default: {
                cout << "Invalid choice!";
                break;
            }

        }

    } while (choice != 7);

    return 0;
}
